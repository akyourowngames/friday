"""CronStore — JSON-based CRUD for cron job definitions with atomic writes."""

from __future__ import annotations

import json
import os
import re
import tempfile
from datetime import datetime, timezone
from pathlib import Path

from croniter import croniter


def _slugify(name: str) -> str:
    """Convert a job name to a kebab-case ID."""
    slug = name.lower().strip()
    slug = re.sub(r"[^a-z0-9\s-]", "", slug)
    slug = re.sub(r"[\s]+", "-", slug)
    slug = re.sub(r"-+", "-", slug).strip("-")
    return slug or "job"


class CronStore:
    """JSON file-backed store for cron job definitions."""

    def __init__(self, data_dir: str | Path) -> None:
        self._data_dir = Path(data_dir).expanduser()
        self._cron_dir = self._data_dir / "cron"
        self._cron_dir.mkdir(parents=True, exist_ok=True)
        self._jobs_file = self._cron_dir / "jobs.json"
        if not self._jobs_file.exists():
            self._write({"jobs": {}})

    def _jobs_path(self) -> Path:
        return self._jobs_file

    def _read(self) -> dict:
        try:
            return json.loads(self._jobs_file.read_text(encoding="utf-8"))
        except (json.JSONDecodeError, FileNotFoundError):
            return {"jobs": {}}

    def _write(self, data: dict) -> None:
        """Atomic write: write to temp file, then rename."""
        fd, tmp_path = tempfile.mkstemp(
            dir=str(self._cron_dir), suffix=".tmp"
        )
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as f:
                json.dump(data, f, indent=2, ensure_ascii=False)
            os.replace(tmp_path, str(self._jobs_file))
        except Exception:
            # Clean up temp file on failure
            try:
                os.unlink(tmp_path)
            except OSError:
                pass
            raise

    def _next_run_at(self, cron_expr: str, timezone_str: str = "UTC") -> str:
        """Compute the next run time for a cron expression."""
        from datetime import datetime, timezone
        now = datetime.now(timezone.utc)
        cron = croniter(cron_expr, now)
        next_dt = cron.get_next(datetime)
        return next_dt.isoformat()

    def create_job(
        self,
        name: str,
        prompt: str,
        cron: str,
        timezone: str = "UTC",
        max_iterations: int = 10,
    ) -> dict:
        """Create a new cron job. Raises ValueError if name already exists."""
        # Validate cron expression
        if not croniter.is_valid(cron):
            raise ValueError(f"Invalid cron expression: {cron}")

        job_id = _slugify(name)

        data = self._read()
        if job_id in data["jobs"]:
            raise ValueError(f"Job with ID '{job_id}' already exists")

        now_iso = datetime.now(timezone.utc).isoformat()
        next_run = self._next_run_at(cron, timezone)

        job = {
            "id": job_id,
            "name": name,
            "prompt": prompt,
            "cron": cron,
            "timezone": timezone,
            "created_at": now_iso,
            "enabled": True,
            "state": "scheduled",
            "next_run_at": next_run,
            "last_run_at": None,
            "run_count": 0,
            "last_status": None,
            "max_iterations": max_iterations,
            "output_dir": str(self._cron_dir / "logs" / job_id),
        }
        data["jobs"][job_id] = job
        self._write(data)

        # Ensure log directory exists
        self.log_dir(job_id).mkdir(parents=True, exist_ok=True)
        return job

    def list_jobs(self, include_disabled: bool = True) -> list[dict]:
        """Return all jobs, optionally filtering out disabled ones."""
        data = self._read()
        jobs = list(data["jobs"].values())
        if not include_disabled:
            jobs = [j for j in jobs if j.get("enabled", True)]
        return jobs

    def get_job(self, job_id: str) -> dict | None:
        """Return a single job by ID, or None."""
        data = self._read()
        return data["jobs"].get(job_id)

    def update_job(self, job_id: str, **fields) -> dict:
        """Update fields on a job. Returns the updated job."""
        data = self._read()
        if job_id not in data["jobs"]:
            raise ValueError(f"Job '{job_id}' not found")
        job = data["jobs"][job_id]
        job.update(fields)
        data["jobs"][job_id] = job
        self._write(data)
        return job

    def delete_job(self, job_id: str) -> None:
        """Delete a job (logs are kept). Raises ValueError if not found."""
        data = self._read()
        if job_id not in data["jobs"]:
            raise ValueError(f"Job '{job_id}' not found")
        del data["jobs"][job_id]
        self._write(data)

    def get_due_jobs(self) -> list[dict]:
        """Return all jobs that are due to run (next_run_at <= now, enabled, not running)."""
        from datetime import datetime, timezone
        now = datetime.now(timezone.utc)
        data = self._read()
        due = []
        for job in data["jobs"].values():
            if not job.get("enabled", True):
                continue
            if job.get("state") == "running":
                continue
            next_run = job.get("next_run_at")
            if not next_run:
                continue
            try:
                next_dt = datetime.fromisoformat(next_run)
                if next_dt.tzinfo is None:
                    next_dt = next_dt.replace(tzinfo=timezone.utc)
                if next_dt <= now:
                    due.append(job)
            except (ValueError, TypeError):
                continue
        # Sort by how overdue they are
        due.sort(key=lambda j: j.get("next_run_at", ""))
        return due

    def log_dir(self, job_id: str) -> Path:
        """Return the log directory for a job, creating it if needed."""
        log_dir = self._cron_dir / "logs" / job_id
        log_dir.mkdir(parents=True, exist_ok=True)
        return log_dir

    def list_log_files(self, job_id: str) -> list[Path]:
        """Return sorted list of log files for a job (oldest first)."""
        log_dir = self.log_dir(job_id)
        files = sorted(log_dir.glob("*.md"))
        return files

    def get_previous_summary(self, job_id: str) -> str | None:
        """Extract the Summary section from the most recent log file."""
        files = self.list_log_files(job_id)
        if not files:
            return None
        last_log = files[-1]
        content = last_log.read_text(encoding="utf-8")
        # Extract between "## Summary" and next "##" or end
        match = re.search(
            r"## Summary\n(.*?)(?=\n## |\Z)",
            content,
            re.DOTALL,
        )
        if match:
            summary = match.group(1).strip()
            return summary if summary else None
        return None
```

"""Cron schedule utilities — validation, natural language parsing, formatting."""

from __future__ import annotations

import re

from croniter import croniter


def validate_cron(expression: str) -> bool:
    """Check if a cron expression is valid."""
    if not expression or not expression.strip():
        return False
    return croniter.is_valid(expression.strip())


def parse_natural_schedule(text: str) -> str | None:
    """Convert natural language to a cron expression. Returns None if unrecognized."""
    text = text.strip().lower()

    # "every N minutes"
    m = re.match(r"every\s+(\d+)\s+minutes?", text)
    if m:
        n = int(m.group(1))
        if n >= 1 and n <= 59:
            return f"*/{n} * * * *"

    # "every N hours"
    m = re.match(r"every\s+(\d+)\s+hours?", text)
    if m:
        n = int(m.group(1))
        if n >= 1 and n <= 23:
            return f"0 */{n} * * *"

    # "every hour"
    if re.match(r"every\s+hour$", text):
        return "0 * * * *"

    # "every day at H:MMam/pm"
    m = re.match(r"every\s+day\s+at\s+(\d{1,2}):(\d{2})\s*(am|pm)", text)
    if m:
        hour = int(m.group(1))
        minute = int(m.group(2))
        period = m.group(3)
        if period == "pm" and hour != 12:
            hour += 12
        elif period == "am" and hour == 12:
            hour = 0
        return f"{minute} {hour} * * *"

    # "every day at Ham/pm"
    m = re.match(r"every\s+day\s+at\s+(\d{1,2})\s*(am|pm)", text)
    if m:
        hour = int(m.group(1))
        period = m.group(2)
        if period == "pm" and hour != 12:
            hour += 12
        elif period == "am" and hour == 12:
            hour = 0
        return f"0 {hour} * * *"

    # "every weekday at H:MMam/pm"
    m = re.match(r"every\s+weekday[s]?\s+at\s+(\d{1,2}):(\d{2})\s*(am|pm)", text)
    if m:
        hour = int(m.group(1))
        minute = int(m.group(2))
        period = m.group(3)
        if period == "pm" and hour != 12:
            hour += 12
        elif period == "am" and hour == 12:
            hour = 0
        return f"{minute} {hour} * * 1-5"

    # "every weekday at Ham/pm"
    m = re.match(r"every\s+weekday[s]?\s+at\s+(\d{1,2})\s*(am|pm)", text)
    if m:
        hour = int(m.group(1))
        period = m.group(2)
        if period == "pm" and hour != 12:
            hour += 12
        elif period == "am" and hour == 12:
            hour = 0
        return f"0 {hour} * * 1-5"

    # "every DAYNAME at H:MMam/pm"
    day_map = {
        "monday": 1, "tuesday": 2, "wednesday": 3,
        "thursday": 4, "friday": 5, "saturday": 6, "sunday": 0,
    }
    for day_name, day_num in day_map.items():
        # with minutes
        m = re.match(
            rf"every\s+{day_name}\s+at\s+(\d{{1,2}}):(\d{{2}})\s*(am|pm)", text
        )
        if m:
            hour = int(m.group(1))
            minute = int(m.group(2))
            period = m.group(3)
            if period == "pm" and hour != 12:
                hour += 12
            elif period == "am" and hour == 12:
                hour = 0
            return f"{minute} {hour} * * {day_num}"

        # without minutes
        m = re.match(
            rf"every\s+{day_name}\s+at\s+(\d{{1,2}})\s*(am|pm)", text
        )
        if m:
            hour = int(m.group(1))
            period = m.group(2)
            if period == "pm" and hour != 12:
                hour += 12
            elif period == "am" and hour == 12:
                hour = 0
            return f"0 {hour} * * {day_num}"

    # "every Nth of the month"
    m = re.match(r"every\s+(\d{1,2})(?:st|nd|rd|th)\s+of\s+the\s+month", text)
    if m:
        day = int(m.group(1))
        if 1 <= day <= 31:
            return f"0 9 {day} * *"

    return None


def format_cron_human(expression: str) -> str:
    """Convert a cron expression to a human-readable string."""
    if not validate_cron(expression):
        return expression

    parts = expression.strip().split()
    if len(parts) < 5:
        return expression

    minute, hour, day, month, dow = parts[0], parts[1], parts[2], parts[3], parts[4]

    # Every N minutes
    if minute.startswith("*/") and hour == "*":
        n = minute[2:]
        return f"Every {n} minutes"

    # Every N hours
    if hour.startswith("*/") and minute == "0":
        n = hour[2:]
        return f"Every {n} hours"

    # Every hour
    if hour == "*" and minute == "0":
        return "Every hour"

    # Specific time patterns
    day_names = {0: "Sunday", 1: "Monday", 2: "Tuesday", 3: "Wednesday",
                 4: "Thursday", 5: "Friday", 6: "Saturday"}

"""CronRunner — creates fresh Agent sessions and writes markdown logs."""

from __future__ import annotations

import time
import traceback
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from ares.cron.store import CronStore


class CronRunner:
    """Runs a cron job as a fresh agent session and saves a markdown log."""

    def __init__(self, store: CronStore) -> None:
        self.store = store

    def _build_run_prompt(self, job: dict) -> str:
        """Build the prompt for a cron run, prepending previous run summary."""
        base_prompt = job["prompt"]
        previous_summary = self.store.get_previous_summary(job["id"])
        if previous_summary:
            return (
                f"## Previous Run Summary\n{previous_summary}\n\n"
                f"## Current Task\n{base_prompt}"
            )
        return base_prompt

    def _save_log(
        self,
        job: dict,
        status: str,
        output: str,
        duration_s: float,
        tokens: int,
        iterations: int,
        tools_used: list[tuple[str, int]],
        error: str | None = None,
    ) -> Path:
        """Save a markdown log for this run. Returns the log file path."""
        now = datetime.now(timezone.utc)
        timestamp = now.strftime("%Y-%m-%dT%H-%M-%SZ")
        log_dir = self.store.log_dir(job["id"])
        log_file = log_dir / f"{timestamp}.md"

        # Get previous summary for the header
        previous_summary = self.store.get_previous_summary(job["id"])

        tools_section = ""
        if tools_used:
            tools_section = "\n".join(
                f"- {name} ({count} call{'s' if count > 1 else ''})"
                for name, count in tools_used
            )
        else:
            tools_section = "No tools used."

        error_section = ""
        if error:
            error_section = f"\n\n## Error\n```\n{error}\n```"

        # Format duration nicely
        if duration_s < 60:
            duration_str = f"{duration_s:.1f}s"
        else:
            mins = int(duration_s // 60)
            secs = duration_s % 60
            duration_str = f"{mins}m {secs:.0f}s"

        content = f"""# Cron Run: {job['name']}
**Job:** {job['id']}
**Run:** {now.isoformat()}
**Status:** {'completed' if status == 'completed' else 'failed'}
**Duration:** {duration_str}
**Tokens:** {tokens:,}

## Previous Run Summary
{previous_summary if previous_summary else '_No previous run._'}

## Prompt
{job['prompt']}

## Agent Output
{output if output else '_No output._'}

## Summary
{output[:500] if output else ('Error: ' + (error or 'Unknown error'))}

## Tools Used
{tools_section}

## Run Metadata
- Model: cron-session
- Iterations: {iterations}/{job.get('max_iterations', 10)}{error_section}
"""
        log_file.write_text(content, encoding="utf-8")
        return log_file

    async def run_job(self, job: dict, agent_factory: Any) -> dict:
        """Execute a cron job. Returns the run result dict.

        agent_factory: callable that returns a fresh Agent instance.
        """
        start_time = time.monotonic()
        error = None
        output = ""
        iterations = 0
        tokens = 0
        tools_used: list[tuple[str, int]] = []

        try:
            # Build prompt with previous run context
            prompt = self._build_run_prompt(job)

            # Create fresh agent session
            agent = agent_factory(is_cron_session=True)

            # Collect all output from the agent
            output_parts = []
            tool_counts: dict[str, int] = {}

            async for chunk in agent.run_stream(prompt, conversation_history=[]):
                if chunk.startswith("[tool:"):
                    # Parse tool usage
                    # Format: [tool:name:result]
                    tool_name = chunk.split("[tool:")[1].split(":")[0]
                    tool_counts[tool_name] = tool_counts.get(tool_name, 0) + 1
                else:
                    output_parts.append(chunk)

            output = "".join(output_parts)
            tools_used = list(tool_counts.items())

            # Count iterations (approximate from tool usage)
            iterations = sum(tool_counts.values()) + 1

            # Get token count from agent if available
            if hasattr(agent, "last_token_count"):
                tokens = agent.last_token_count

        except Exception:
            error = traceback.format_exc()

        duration_s = time.monotonic() - start_time
        status = "completed" if error is None else "failed"

        # Save the log
        log_path = self._save_log(
            job=job,
            status=status,
            output=output,
            duration_s=duration_s,
            tokens=tokens,
            iterations=iterations,
            tools_used=tools_used,
            error=error,
        )

        # Compute next run time
        from croniter import croniter
        now = datetime.now(timezone.utc)
        cron = croniter(job["cron"], now)
        next_dt = cron.get_next(datetime)
        next_run_at = next_dt.isoformat()

        return {
            "status": status,
            "duration_s": duration_s,
            "log_path": str(log_path),
            "next_run_at": next_run_at,
            "error": error,
        }
```

"""CronScheduler — asyncio background task that ticks and runs due jobs."""

from __future__ import annotations

import asyncio
import logging
from typing import Any, Callable

from ares.cron.runner import CronRunner
from ares.cron.store import CronStore

logger = logging.getLogger(__name__)


class CronScheduler:
    """Background scheduler that ticks every N seconds and runs due jobs."""

    def __init__(
        self,
        store: CronStore,
        runner: CronRunner | None = None,
        agent_factory: Callable[..., Any] | None = None,
        tick_seconds: int = 60,
        max_concurrent: int = 3,
    ) -> None:
        self.store = store
        self.runner = runner or CronRunner(store=store)
        self.agent_factory = agent_factory
        self.tick_seconds = tick_seconds
        self.max_concurrent = max_concurrent
        self._task: asyncio.Task | None = None
        self._running_jobs: dict[str, asyncio.Task] = {}

    async def start(self) -> None:
        """Start the background tick loop."""
        if self._task is not None:
            return
        self._task = asyncio.create_task(self._run_loop())
        logger.info("CronScheduler started (tick=%ds)", self.tick_seconds)

    async def stop(self) -> None:
        """Stop the tick loop and wait for running jobs to finish."""
        if self._task is None:
            return
        self._task.cancel()
        try:
            await self._task
        except asyncio.CancelledError:
            pass
        self._task = None

        # Wait for any running jobs
        if self._running_jobs:
            await asyncio.gather(
                *self._running_jobs.values(),
                return_exceptions=True,
            )
        logger.info("CronScheduler stopped")

    async def _run_loop(self) -> None:
        """Main tick loop."""
        while True:
            try:
                await self._tick()
            except asyncio.CancelledError:
                raise
            except Exception:
                logger.exception("CronScheduler tick error")
            await asyncio.sleep(self.tick_seconds)

    async def _tick(self) -> None:
        """One tick: find due jobs and run them."""
        due_jobs = self.store.get_due_jobs()
        if not due_jobs:
            return

        # Respect max concurrency
        slots = self.max_concurrent - len(self._running_jobs)
        if slots <= 0:
            return

        for job in due_jobs[:slots]:
            await self._run_job(job)

    async def _run_job(self, job: dict) -> None:
        """Run a single job in the background."""
        job_id = job["id"]

        # Mark as running
        self.store.update_job(job_id, state="running")

        async def _execute():
            try:
                result = await self.runner.run_job(job, self.agent_factory)
                # Update job state after completion
                self.store.update_job(
                    job_id,
                    state="scheduled",
                    last_run_at=result.get("next_run_at"),
                    run_count=job.get("run_count", 0) + 1,
                    last_status=result["status"],
                    next_run_at=result["next_run_at"],
                )
            except Exception:
                logger.exception("Cron job %s failed", job_id)
                self.store.update_job(
                    job_id,
                    state="scheduled",
                    last_status="failed",
                )
            finally:
                self._running_jobs.pop(job_id, None)

        task = asyncio.create_task(_execute())
        self._running_jobs[job_id] = task
```

            "run_cron_job_now": self._run_now,
            "get_cron_logs": self._get_logs,
        }
        handler = handlers.get(tool_name)
        if handler is None:
            raise ValueError(f"Unknown cron tool: {tool_name}")
        return handler(args)

    def _create(self, args: dict) -> str:
        name = args["name"]
        prompt = args["prompt"]
        cron_expr = args["cron"]
        timezone = args.get("timezone", "UTC")

        # Try natural language parsing if cron looks invalid
        if not validate_cron(cron_expr):
            parsed = parse_natural_schedule(cron_expr)
            if parsed is None:
                raise ValueError(
                    f"Invalid cron expression: '{cron_expr}'. "
                    f"Use standard cron format like '0 9 * * *' or natural language like 'every day at 9am'."
                )
            cron_expr = parsed

        job = self.store.create_job(
            name=name,
            prompt=prompt,
            cron=cron_expr,
            timezone=timezone,
        )
        return json.dumps(job, indent=2)

    def _list(self, args: dict) -> str:
        include_disabled = args.get("include_disabled", True)
        jobs = self.store.list_jobs(include_disabled=include_disabled)
        return json.dumps({
            "count": len(jobs),
            "jobs": jobs,
        }, indent=2)

    def _get(self, args: dict) -> str:
        job_id = args["job_id"]
        job = self.store.get_job(job_id)
        if job is None:
            raise ValueError(f"Job '{job_id}' not found")
        return json.dumps(job, indent=2)

    def _update(self, args: dict) -> str:
        job_id = args["job_id"]
        fields = {}
        for key in ("name", "prompt", "cron", "timezone", "enabled"):
            if key in args:
                fields[key] = args[key]

        # Validate cron if being updated
        if "cron" in fields and not validate_cron(fields["cron"]):
            parsed = parse_natural_schedule(fields["cron"])
            if parsed is None:
                raise ValueError(f"Invalid cron expression: '{fields['cron']}'")
            fields["cron"] = parsed

        job = self.store.update_job(job_id, **fields)
        return json.dumps(job, indent=2)

    def _delete(self, args: dict) -> str:
        job_id = args["job_id"]
        job = self.store.get_job(job_id)
        if job is None:
            raise ValueError(f"Job '{job_id}' not found")
        self.store.delete_job(job_id)
        return json.dumps({"message": f"Job '{job_id}' deleted", "job_id": job_id})

    def _run_now(self, args: dict) -> str:
        job_id = args["job_id"]
        job = self.store.get_job(job_id)
        if job is None:
            raise ValueError(f"Job '{job_id}' not found")
        # Trigger immediate run by setting next_run_at to now
        from datetime import datetime, timezone as tz
        now = datetime.now(tz.utc).isoformat()
        self.store.update_job(job_id, next_run_at=now, state="scheduled")
        return json.dumps({"message": f"Job '{job_id}' triggered", "status": "triggered"})

    def _get_logs(self, args: dict) -> str:
        job_id = args["job_id"]
        limit = args.get("limit", 5)
        job = self.store.get_job(job_id)
        if job is None:
            raise ValueError(f"Job '{job_id}' not found")

        log_files = self.store.list_log_files(job_id)
        # Return most recent N
        recent = log_files[-limit:] if len(log_files) > limit else log_files

        logs = []
        for log_file in recent:
            content = log_file.read_text(encoding="utf-8")
            logs.append({
                "file": log_file.name,
                "content": content,
            })

        return json.dumps({
            "job_id": job_id,
            "count": len(logs),
            "logs": logs,
        }, indent=2)
```

        """Delete log files older than max_age_days. Returns count deleted."""
        from datetime import timedelta
        cutoff = datetime.now(timezone.utc) - timedelta(days=max_age_days)
        deleted = 0
        for job_dir in (self._cron_dir / "logs").iterdir():
            if not job_dir.is_dir():

            for log_file in job_dir.glob("*.md"):
                # Parse timestamp from filename
                try:
                    ts_str = log_file.stem  # e.g. "2026-06-28T09-00-00Z"
                    file_dt = datetime.fromisoformat(ts_str.replace("Z", "+00:00"))
                    if file_dt < cutoff:
                        log_file.unlink()
                        deleted += 1
                except (ValueError, OSError):
                    continue
        return deleted
